// acwing252 点分治
#include<bits/stdc++.h>

using namespace std;
const int N = 1e4 + 10;
struct node{
    int v, w;
};
vector<node> g[N];
bool vis[N];
long long ans = 0;
int siz[N], col[N], dis[N], cnt[N], a[N], tot;
// col[u] : u在pos中的哪颗子树（颜色）
// dis[u] : u到pos的距离
// cnt[x] : 颜色为x的有几个
// a[i] : 按dis排序后的数组
int pos, max_part;
int n, k; 
void dfs_find(int u, int fa, int S){
    // 找重心(pos)
    siz[u] = 1;
    int mx = 1;
    for(auto x : g[u]){
        if(x.v == fa || vis[x.v]) continue;
        dfs_find(x.v, u, S);
        siz[u] += siz[x.v];
        mx = max(mx, siz[x.v]);
    }
    mx = max(mx, S - siz[u]);
    if(mx < max_part){
        max_part = mx;
        pos = u;
    }
}
void dfs(int u, int fa){
    a[++ tot] = u;
    if(fa == pos) col[u] = u;
    cnt[col[u]] ++;
    for(auto x : g[u]){
        if(x.v == fa || vis[x.v]) continue;
        if(col[u]){
            col[x.v] = col[u];
        }
        dis[x.v] = dis[u] + x.w;
        dfs(x.v, u);
    }
}
bool cmp(int x, int y){
    return dis[x] < dis[y];
}
void work(int S, int x){ // tot_siz, v
    max_part = N, dfs_find(x, 0, S);
    for(int i = 0; i <= n; i ++) cnt[i] = 0;
    vis[pos] = 1;
    tot = col[pos] = dis[pos] = 0, dfs(pos, 0);
    sort(a + 1, a + 1 + tot, cmp);
    for(int l = 1, r = tot; l <= r; ){
        while(l < r && dis[a[r]] + dis[a[l]] > k){ // 删r
            cnt[col[a[r--]]] --;
        }
        if(dis[a[r]] + dis[a[l]] <= k && l < r) 
            ans += r - l + 1 - cnt[col[a[l]]];
        cnt[col[a[l ++]]] --;
    }
    for(auto y : g[pos]){
        if(!vis[y.v]){
            work(siz[y.v], y.v);
        }
    }
}

void sol(){
    for(int i = 1; i <= n; i ++) g[i].clear(), vis[i] = 0;
    for(int i = 1; i < n; i ++){
        int u, v, w; cin >> u >> v >> w;
        u ++, v ++;
        g[u].push_back({v, w});
        g[v].push_back({u, w});
    }
    ans = 0;
    work(n, 1);
    cout << ans << '\n';
}
int main(){
    // freopen("1.txt", "r", stdin);
    while(cin >> n >> k && n && k){
        sol();
    }
    return 0;
}
